cmake_minimum_required(VERSION 3.16)
project(SupplyChainSupplier LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(supplier-lazy-accessibility STATIC lazy_accessibility.cpp)
target_compile_definitions(supplier-lazy-accessibility PUBLIC UNICODE _UNICODE NOMINMAX)
target_compile_options(supplier-lazy-accessibility PRIVATE /utf-8)
target_link_libraries(supplier-lazy-accessibility PUBLIC comctl32 ole32 oleacc oleaut32 uiautomationcore uuid delayimp)
target_link_options(supplier-lazy-accessibility INTERFACE /DELAYLOAD:UIAutomationCore.dll /DELAYLOAD:OLEACC.dll)

add_executable(supply-chain-supplier WIN32 main.cpp app.rc)
target_compile_definitions(supply-chain-supplier PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(supply-chain-supplier PRIVATE /utf-8)
target_link_libraries(supply-chain-supplier PRIVATE supplier-lazy-accessibility comctl32 comdlg32 winhttp shell32)
set_target_properties(supply-chain-supplier PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../build/win32")
add_executable(supplier-model-tests model_tests.cpp)
target_compile_options(supplier-model-tests PRIVATE /utf-8)
enable_testing()
add_test(NAME supplier-model COMMAND supplier-model-tests)

add_executable(supplier-accessibility-tests lazy_accessibility_tests.cpp)
target_compile_options(supplier-accessibility-tests PRIVATE /utf-8)
target_link_libraries(supplier-accessibility-tests PRIVATE supplier-lazy-accessibility)
add_test(NAME supplier-accessibility COMMAND supplier-accessibility-tests)
set_tests_properties(supplier-accessibility PROPERTIES TIMEOUT 60)

add_executable(supplier-accessibility-probe lazy_accessibility_probe.cpp)
target_compile_definitions(supplier-accessibility-probe PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(supplier-accessibility-probe PRIVATE /utf-8)
target_link_libraries(supplier-accessibility-probe PRIVATE ole32 oleacc oleaut32 uiautomationcore uuid)
target_link_options(supply-chain-supplier PRIVATE /MANIFEST:NO)
set_property(TARGET supply-chain-supplier supplier-model-tests supplier-lazy-accessibility supplier-accessibility-tests supplier-accessibility-probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
